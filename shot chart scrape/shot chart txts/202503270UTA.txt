// Update the shot chart
function update_shot_chart(team_id) {

  var q_elem = document.getElementById('quarter-' + team_id);
  var p_elem = document.getElementById('player-'  + team_id);
  var s_elem = document.getElementById('shots-'   + team_id);

  var quarter = q_elem.options[q_elem.selectedIndex].value;
  var player  = p_elem.options[p_elem.selectedIndex].value;

  for (var i = 1; i < s_elem.childNodes.length; i++) {

    var c_elem = s_elem.childNodes[i];

    if (c_elem.nodeName.toLowerCase() !== 'div') { continue; }

    if ((c_elem.className.indexOf('p-' + player) === -1)
         || (c_elem.className.indexOf('q-' + quarter) === -1)) {

      c_elem.style.visibility = "hidden";
    }
    else {
      c_elem.style.visibility = "visible";
    }
  }
}

function reset_shot_chart(team_id) {

  var q_elem = document.getElementById('quarter-' + team_id);
  var p_elem = document.getElementById('player-'  + team_id);
  var s_elem = document.getElementById('shots-'   + team_id);

  for (var i = 0; i < s_elem.childNodes.length; i++) {

    var c_elem = s_elem.childNodes[i];

    if (c_elem.nodeName.toLowerCase() !== 'div') { continue; }

    c_elem.style.visibility = "visible";
  }

  q_elem.selectedIndex = 0;
  p_elem.selectedIndex = 0;
}
</script>
